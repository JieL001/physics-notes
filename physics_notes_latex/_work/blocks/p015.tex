\textbf{安培力做功}：
\[
\left\{\begin{aligned}
&\text{恒力}\quad \pm BIL\cdot x\\
&\text{变力}\quad \pm W_{\text{安}}
\end{aligned}\right.
\]
\textbf{安培力冲量}：
\[
\left\{\begin{aligned}
&\text{恒力}\quad \pm BILt=\pm BLq\\
&\text{变力}\quad \pm I_{A}=\pm BLq
\end{aligned}\right.
\]
