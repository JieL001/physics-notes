\[
\left\{\begin{aligned}
&N_1\cos\theta+f_1\sin\theta=mg\\
&f_1\cos\theta-N\sin\theta=ma_1 % CHECK: N 无下标，应为 N_1
\\
&a_1=kt_1\\
&f_1=\mu N_1
\end{aligned}\right.
\]
\[
N_1=\frac{mg}{\cos\theta+\mu\sin\theta}\qquad f_1=\frac{\mu mg}{\cos\theta+\mu\sin\theta}\qquad t_1=\frac{g(\mu\cos\theta-\sin\theta)}{k(\mu\sin\theta+\cos\theta)}
\]
\[
\text{斜面对物块}\ I_1=t_1\sqrt{N_1^2+f_1^2}=\frac{mg^2(\mu\cos\theta-\sin\theta)\sqrt{1+\mu^2}}{k(\mu\sin\theta+\cos\theta)^2}
\]
