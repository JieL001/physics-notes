\begin{problem}
$L=3.2\unit{m}$　　$\mu=0.5$　　$m=2\unit{kg}$　　$V_0=2\unit{m/s}$

%FIG p237_a 0.01 0.04 0.23 0.145
\notefig[0.25]{figures/p237_a.png}

\begin{solution}
\begin{gather*}
a_1=g\sin\theta+\mu g\cos\theta=10\unit{m/s^2}\\
t_1=t_{\text{共}}=0.2\unit{s}\qquad S_1=\frac{2}{2}\,0.2=0.2\unit{m}<3.2\unit{m}\\ % CHECK: t 的下标手写像"共/类"，按"共"（共速）转录
\because\tan37^\circ=\frac34>\frac12\\ % CHECK: 起首为点状符号，按 ∵ 转录
a_2=g\sin\theta-\mu g\cos\theta=2\unit{m/s^2}\\
S_2=2t_2+\frac12\,2t_2^2=3\unit{m}\\
t^2+2t-3=0\\
(t+3)(t-1)=0\\
t_2=1\unit{s}\\
t=t_1+t_2=1.2\unit{s}\\
V_{\text{末}}=2+1\times2=4\unit{m/s}
\end{gather*}

%FIG p237_b 0.0 0.29 0.32 0.427
\notefig[0.3]{figures/p237_b.png}

\begin{gather*}
S_{\text{相}}=\frac12\,2\times0.2+\frac12\,2\,1=1.2\unit{m}\\
Q=\mu mg\cos\theta\cdot S_{\text{相}}=9.6\unit{J}
\end{gather*}
（$Q$ 行下小字：\unclear{4　8-}）
\end{solution}

%FIG p237_c 0.72 0.463 0.835 0.527

\end{array}\right.
\] % CHECK: 第二行首字手写像 Y/S，按 S（痕迹长度）转录
\end{problem}
