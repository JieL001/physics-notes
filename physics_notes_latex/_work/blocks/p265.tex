%FIG p265_a 0.00 0.00 0.21 0.085

\begin{align*}
&V_x=V_0\\
&V_y=gt\\
&V=\sqrt{V_x^2+V_y^2}
\end{align*}
\[ \tan\theta=\dfrac{V_0}{V_y}=\dfrac{V_0}{gt}\qquad t=\dfrac{V_0}{g\tan\theta} \]

\textbf{抛上斜面}
%FIG p265_d 0.00 0.21 0.26 0.285
\notefig[0.3]{figures/p265_d.png}
\[
\left\{\begin{aligned}
x&=V_0t\\
y&=\dfrac12gt^2\\
x_{\text{合}}&=\sqrt{x^2+y^2}
\end{aligned}\right.
\qquad
\begin{aligned}
\tan\theta&=\dfrac{y}{x}=\dfrac{gt}{2V_0}\\
t&=\dfrac{2V_0\tan\theta}{g}
\end{aligned}
\]

\textbf{三类光滑斜面下滑时间比较}
\begin{itemize}
\item \textbf{等高斜面}\quad $L=\frac12at^2$\quad $a=g\sin\theta$\quad $L=\dfrac{h}{\sin\theta}$

$t=\dfrac{1}{\sin\theta}\sqrt{\dfrac{2h}{g}}$

$\theta$ 越小 时间越长
%FIG p265_h 0.05 0.924 0.195 0.972
\notefig[0.25]{figures/p265_h.png}
$t_1>t_2>t_3$
\item \textbf{同底斜面}\quad $L=\frac12at^2$\quad $a=g\sin\theta$\quad $L=\dfrac{d}{\cos\theta}$

$t=\sqrt{\dfrac{4d}{g\sin2\theta}}$
%FIG p265_i 0.51 0.848 0.64 0.906
\notefig[0.25]{figures/p265_i.png}
$\theta=45^\circ$ 时 $t_{\min}$

关于 $45^\circ$ 对称 $t$ 相等
%FIG p265_j 0.33 0.952 0.41 1.0
\notefig[0.25]{figures/p265_j.png}
$t_1=t_3>t_2$
\item \textbf{同圆周的斜面}——等时圆
\begin{itemize}
\item 同顶：$\sin\theta\,2R=\frac12g\sin\theta\,t^2$\quad $t_1=t_2=t_3$
\item 同底\quad $t_1=t_2=t_3$
\item 双圆内斜面\quad $t_1=t_2=t_3$
\end{itemize}
%FIG p265_k 0.59 0.918 0.865 1.0
\notefig[0.4]{figures/p265_k.png}
\end{itemize}
