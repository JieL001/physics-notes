\begin{problem}
%FIG p223_b 0.07 0.15 0.225 0.20
\notefig[0.25]{figures/p223_b.png}
$\leftarrow F_2$ 用 $t$ 后回原地
\begin{solution}
以向右为正

$\dfrac{F_1}{F_2}=\dfrac13$

%FIG p223_c 0.515 0.13 0.775 0.305
\notefig[0.3]{figures/p223_c.png}
\begin{align*}
a_1&=\frac{F_1}{m}\qquad x_1=\frac12\frac{F_1}{m}t^2\\
V_1&=a_1t=\frac{F_1t}{m}
\end{align*}
第2个$t$中。$a_2=\dfrac{-F_2}{m}$
\begin{align*}
x_2&=V_1t+\frac12a_2t^2\\
x_2&=\frac{F_1t}{m}+\frac12\left(-\frac{F_2}{m}\right)t^2 % CHECK: 第一项分子原稿为 F1t（按 V1t 应为 F1t^2/m）
\end{align*}
又在 $0\sim2t$ 内 $x_1+x_2=0$\quad $\therefore\ \dfrac{F_1}{F_2}=\dfrac13$
\end{solution}
\end{problem}
