\providecommand\notefigroot{}   % 编译目录不是项目根目录时（如 texcheck）由外部定义为项目路径
\newif\ifnotefigtikz \notefigtikztrue
\makeatletter
\newsavebox\notefig@box
\def\notefig@name figures/#1.png\@nil{#1}
\newcommand{\notefig}[2][0.6]{\begin{center}\notefig@do{#1}{#2}\end{center}}
\def\notefig@do#1#2{%
  \edef\notefig@base{\notefig@name#2\@nil}%
  \ifnotefigtikz
    \IfFileExists{\notefigroot figures_tikz/\notefig@base.tex}{\notefig@tikz}{\notefig@png{#1}{#2}}%
  \else\notefig@png{#1}{#2}\fi}
\def\notefig@png#1#2{\includegraphics[width=#1\linewidth,height=0.45\textheight,keepaspectratio]{#2}}
\def\notefig@tikz{%
  \sbox\notefig@box{\small\input{\notefigroot figures_tikz/\notefig@base.tex}}%
  \ifdim\wd\notefig@box>\linewidth\relax\resizebox{\linewidth}{!}{\usebox\notefig@box}%
  \else\ifdim\ht\notefig@box>0.5\textheight\relax\resizebox{!}{0.5\textheight}{\usebox\notefig@box}%
  \else\usebox\notefig@box\fi\fi}
\makeatother
