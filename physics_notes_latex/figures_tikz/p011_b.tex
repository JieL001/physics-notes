\begin{tikzpicture}[line width=0.7pt, >={Stealth[length=2.2mm]}]
  \coordinate (A) at (0,2.8);
  \coordinate (B) at (2,1.4);
  \coordinate (C) at (2,-1.4);
  \coordinate (D) at (0,0);
  \draw (A) -- (B) -- (C) -- (D) -- cycle;
  \node[above left, inner sep=2pt] at (A) {$A$};
  \node[right, inner sep=3pt] at (B) {$B$};
  \node[below right, inner sep=2pt] at (C) {$C$};
  \node[below left, inner sep=2pt] at (D) {$D$};
\end{tikzpicture}
