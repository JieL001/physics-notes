% p016_a 的电脑重画（平移圆：V0 方向不变，入射位置改变；两个等大的圆，第二个圆的下半加粗）
\begin{tikzpicture}[line width=0.7pt, >={Stealth[length=2.2mm]}]
  % 两个等大的圆（淡）
  \draw[gray!55] (1.1,0) circle (1.1);
  \draw[gray!55] (2.5,0) circle (1.1);
  % 磁场（点）
  \foreach \x in {0.4,1.2,2.0} {
    \foreach \y in {1.2,0.25,-0.65} {
      \fill (\x,\y) circle (1.1pt);
    }
  }
  % 边界线
  \draw[thick] (0,0) -- (6.6,0);
  % 第二个圆的轨迹（加粗）
  \draw[line width=1.2pt] (1.4,0) arc (180:380:1.1);
  % 两处入射位置的速度 V0（向上）
  \draw[->, thick] (0,0) -- (0,0.9);
  \draw[->, thick] (1.4,0) -- (1.4,0.9);
\end{tikzpicture}
