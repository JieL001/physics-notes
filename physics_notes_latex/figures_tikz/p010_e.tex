% p010_e 的电脑重画（两幅带电球体示意图：上为同性（电荷在两球外侧），下为异性（电荷在两球相对一侧）；球心连线标 r）
\begin{tikzpicture}[line width=0.7pt, >={Stealth[length=2.2mm]}]
  % 上图：同性（均为 +，分布在远离对方的一侧）
  \draw (0,0) circle (0.6);
  \draw (2.3,0) circle (0.6);
  \draw (0,0) -- (2.3,0);
  \fill (0,0) circle (0.04);
  \node[above] at (1.15,0) {$r$};
  \foreach \y in {0.4,0,-0.4} {
    \draw (-0.92,\y) +(-0.07,0) -- +(0.07,0) (-0.92,\y) +(0,-0.07) -- +(0,0.07);
    \draw (3.22,\y)  +(-0.07,0) -- +(0.07,0) (3.22,\y)  +(0,-0.07) -- +(0,0.07);
  }
  % 下图：异性（左球为 −，右球为 +，均在相对的一侧）
  \begin{scope}[shift={(0,-2.1)}]
    \draw (0,0) circle (0.6);
    \draw (2.3,0) circle (0.6);
    \draw (0,0) -- (2.3,0);
    \node[above] at (1.15,0) {$r$};
    \foreach \y in {0.4,0,-0.4} {
      \draw (0.3,\y) +(-0.07,0) -- +(0.07,0);
      \draw (1.97,\y) +(-0.07,0) -- +(0.07,0) (1.97,\y) +(0,-0.07) -- +(0,0.07);
    }
  \end{scope}
\end{tikzpicture}
